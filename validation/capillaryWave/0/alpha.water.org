// Lower fluid (alpha = 1) below y = a0 cos(2 pi x), a0 = 0.01;
// tanh profile of width 1.5 h, averaged over the cell height (the
// amplitude is a fraction of the cell size)
dimensions      [0 0 0 0 0 0 0];
internalField   #codeStream
{
    codeInclude #{ #include "fvCFD.H" #};
    codeOptions #{ -I$(LIB_SRC)/finiteVolume/lnInclude -I$(LIB_SRC)/meshTools/lnInclude #};
    codeLibs    #{ -lmeshTools -lfiniteVolume #};
    code
    #{
        const IOdictionary& d = static_cast<const IOdictionary&>(dict);
        const fvMesh& mesh = refCast<const fvMesh>(d.db());
        const scalar h = mesh.bounds().span().z();      // dz = h
        const scalar pi = constant::mathematical::pi;
        scalarField alpha(mesh.nCells(), 0.0);
        // cell average in y of 0.5*(1 - tanh(b (y - eta))), b = pi/(1.5 h):
        // int tanh(b s) ds = ln(cosh(b s))/b
        const scalar b = pi/(1.5*h);
        auto lncosh = [](const scalar s)
        {
            return mag(s) + std::log1p(Foam::exp(-2*mag(s))) - Foam::log(2.0);
        };
        forAll(alpha, i)
        {
            const vector& c = mesh.C()[i];
            const scalar eta = 0.01*Foam::cos(2*pi*c.x());
            const scalar s1 = c.y() + 0.5*h - eta;
            const scalar s0 = c.y() - 0.5*h - eta;
            alpha[i] = 0.5*(1.0 - (lncosh(b*s1) - lncosh(b*s0))/(b*h));
        }
        alpha.writeEntry("", os);
    #};
};
boundaryField
{
    #includeEtc "caseDicts/setConstraintTypes"
    "(bottom|top)" { type zeroGradient; }
}
