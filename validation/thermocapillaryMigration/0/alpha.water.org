FoamFile
{
    version     2.0;
    format      ascii;
    class       volScalarField;
    object      alpha.water;
}

// Drop (alpha = 1) of radius 0.5 at the origin, tanh profile of width 1.5 h
dimensions      [0 0 0 0 0 0 0];
internalField   #codeStream
{
    codeInclude #{ #include "fvCFD.H" #};
    codeOptions #{ -I$(LIB_SRC)/finiteVolume/lnInclude -I$(LIB_SRC)/meshTools/lnInclude #};
    codeLibs    #{ -lmeshTools -lfiniteVolume #};
    code
    #{
        const IOdictionary& d = static_cast<const IOdictionary&>(dict);
        const fvMesh& mesh = refCast<const fvMesh>(d.db());
        // uniform mesh, cells ordered radially first: h = radial spacing
        const scalar h = mag(mesh.C()[1].x() - mesh.C()[0].x());
        const scalar pi = constant::mathematical::pi;
        scalarField alpha(mesh.nCells(), 0.0);
        forAll(alpha, i)
        {
            const vector& c = mesh.C()[i];
            const scalar r = Foam::sqrt(sqr(c.x()) + sqr(c.y()) + sqr(c.z()));
            alpha[i] = 0.5*(1.0 - Foam::tanh(pi/(1.5*h)*(r - 0.5)));
        }
        alpha.writeEntry("", os);
    #};
};
boundaryField
{
    "(side|top|bottom)" { type zeroGradient; }
    "(front|back)"      { type wedge; }
    axis                { type empty; }
}
