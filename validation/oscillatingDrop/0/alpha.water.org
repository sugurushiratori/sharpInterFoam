// Drop (alpha = 1) r(theta) = R (1 + e P2(cos theta)), R = 1, e = 0.05,
// theta from the z axis; tanh profile of width 1.5 h
dimensions      [0 0 0 0 0 0 0];
internalField   #codeStream
{
    codeInclude #{ #include "fvCFD.H" #};
    codeOptions #{ -I$(LIB_SRC)/finiteVolume/lnInclude -I$(LIB_SRC)/meshTools/lnInclude #};
    codeLibs    #{ -lmeshTools -lfiniteVolume #};
    code
    #{
        const IOdictionary& d = static_cast<const IOdictionary&>(dict);
        const fvMesh& mesh = refCast<const fvMesh>(d.db());
        // uniform mesh: h = axial spacing of the cell centres
        scalar h = GREAT;
        forAll(mesh.C(), i)
        {
            const scalar dz = mag(mesh.C()[i].z() - mesh.C()[0].z());
            if (dz > SMALL) h = min(h, dz);
        }
        const scalar pi = constant::mathematical::pi;
        scalarField alpha(mesh.nCells(), 0.0);
        forAll(alpha, i)
        {
            const vector& c = mesh.C()[i];
            const scalar rho = Foam::sqrt(sqr(c.x()) + sqr(c.y()));
            const scalar r = Foam::sqrt(sqr(rho) + sqr(c.z()));
            const scalar mu = r > SMALL ? c.z()/r : 1;
            const scalar P2 = 0.5*(3*sqr(mu) - 1);
            const scalar dist = r - (1.0 + 0.05*P2);
            alpha[i] = 0.5*(1.0 - Foam::tanh(pi/(1.5*h)*dist));
        }
        alpha.writeEntry("", os);
    #};
};
boundaryField
{
    #includeEtc "caseDicts/setConstraintTypes"
    "(side|top)"    { type zeroGradient; }
}
